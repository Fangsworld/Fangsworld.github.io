```typescript
// ========== 分类相关 ==========

/**
 * 获取所有分类（树形结构）
 */
export async function getCategoryTree(): Promise<Category[]> {
  const db = getDb();

  // 获取所有分类
  const allCategories = await db.select()
    .from(categories)
    .where(eq(categories.isActive, true))
    .orderBy(categories.sortOrder);

  // 构建树形结构
  const categoryMap = new Map<number, Category>();
  const rootCategories: Category[] = [];

  // 第一遍：创建映射
  allCategories.forEach(cat => {
    categoryMap.set(cat.id, { ...cat, children: [] });
  });

  // 第二遍：构建层级
  allCategories.forEach(cat => {
    const category = categoryMap.get(cat.id)!;
    if (cat.parentId) {
      const parent = categoryMap.get(cat.parentId);
      if (parent) {
        parent.children = parent.children || [];
        parent.children.push(category);
      }
    } else {
      rootCategories.push(category);
    }
  });

  return rootCategories;
}

export async function searchTags(query: string): Promise<Tag[]> {
  const db = getDb();
  return await db.select()
    .from(tags)
    .where(
      and(
        eq(tags.isActive, true),
        or(

          ilike(tags.description, `%${query}%`)

    )
    .orderBy(desc(tags.articleCount))
    .limit(10);

export async function getArticlesByCategory(categorySlug: string): Promise<ArticleMetadata[]> {
  const db = getDb();

  const articles = await db.select({

/**
 * 根据标签获取文章
 */
export async function getArticlesByTag(tagSlug: string): Promise<ArticleMetadata[]> {
  const db = getDb();

  const articles = await db.select({
    id: articles.id,
    title: articles.title,
    excerpt: articles.excerpt,
    date: articles.createdAt,
    slug: articles.slug,
    views: articles.views,

  .from(articles)
  .innerJoin(articleTags, eq(articles.id, articleTags.articleId))
  .innerJoin(tags, eq(articleTags.tagId, tags.id))
  .where(eq(tags.slug, tagSlug))
  .orderBy(desc(articles.createdAt));

  return await enrichArticlesWithRelations(articles);

/**
 * 根据多个标签筛选（AND 逻辑）
 */
export async function getArticlesByTags(tagSlugs: string[]): Promise<ArticleMetadata[]> {
  const db = getDb();

  // 子查询：获取包含所有指定标签的文章 ID
  const matchingArticleIds = await db.select({ articleId: articleTags.articleId })
    .from(articleTags)
    .innerJoin(tags, eq(articleTags.tagId, tags.id))
    .where(inArray(tags.slug, tagSlugs))
    .groupBy(articleTags.articleId)
    .having(sql`COUNT(*) = ${tagSlugs.length}`);

  if (matchingArticleIds.length === 0) return [];

  const articleIds = matchingArticleIds.map(m => m.articleId);

  const articles = await db.select()
    .from(articles)
    .where(inArray(articles.id, articleIds))
    .orderBy(desc(articles.createdAt));

  return await enrichArticlesWithRelations(articles);
}

/**
 * 获取文章的分类和标签
 */
export async function enrichArticlesWithRelations(
  articles: Array<{ id: number; title: string; excerpt: string; date: Date; slug: string; views?: number }>
): Promise<ArticleMetadata[]> {
  const db = getDb();

  const result = await Promise.all(articles.map(async (article) => {
    // 获取分类
    const categories = await db.select({
      id: categories.id,
      slug: categories.slug,
      name: categories.name,
      icon: categories.icon,
      color: categories.color,
    })
    .from(categories)
    .innerJoin(articleCategories, eq(categories.id, articleCategories.categoryId))
    .where(eq(articleCategories.articleId, article.id));

    // 获取主分类
    const primaryCategory = categories.find(c => c.isPrimary) || categories[0];

    // 获取标签
    const tags = await db.select()
    .from(tags)
    .innerJoin(articleTags, eq(tags.id, articleTags.tagId))
    .where(eq(articleTags.articleId, article.id));

    return {
      ...article,
      date: article.date.toISOString(),
      categories,
      primaryCategory,
      tags: tags.map(t => ({
        id: t.tags.id,
        slug: t.tags.slug,
        name: t.tags.name,
        color: t.tags.color,
      })),
    };
  }));

  return result;
}
```

```typescript
// components/articles/TagCloud.tsx

export async function TagCloud() {
  const tags = await getPopularTags(30);

  // 计算字体大小（基于文章数）
  const maxCount = Math.max(...tags.map(t => t.articleCount));
  const minCount = Math.min(...tags.map(t => t.articleCount));

  return (
    <div className="tag-cloud">
      {tags.map(tag => {
        const size = calculateSize(tag.articleCount, minCount, maxCount);
        return (
          <Link
            key={tag.id}
            href={`/tag/${tag.slug}`}
            className="tag"
            style={{
              fontSize: `${size}px`,
              color: tag.color || 'var(--color-text-secondary)',
            }}
          >
            {tag.name}
            <span className="count">({tag.articleCount})</span>
          </Link>
        );
      })}
    </div>
  );
}

function calculateSize(count: number, min: number, max: number): number {
  if (max === min) return 14;
  return 12 + ((count - min) / (max - min)) * 8;
}
```

```typescript
  const [sortBy, setSortBy] = useState<'date' | 'views'>('date');

  return (
    <div className="article-filter">
      {/* 分类选择 */}
      <div className="filter-group">
        <h3>分类</h3>
        <CategorySelector
          multiple
          value={selectedCategories}
          onChange={setSelectedCategories}
        />
      </div>

      {/* 标签选择 */}
      <div className="filter-group">
        <h3>标签</h3>
        <TagInput
          value={selectedTags}
          onChange={setSelectedTags}
          placeholder="搜索标签..."
        />
      </div>

      {/* 时间范围 */}
      <div className="filter-group">
        <h3>时间</h3>
        <DateRangePicker value={dateRange} onChange={setDateRange} />
      </div>

      {/* 排序 */}
      <div className="filter-group">
        <h3>排序</h3>
        <select value={sortBy} onChange={(e) => setSortBy(e.target.value)}>
          <option value="date">最新发布</option>
          <option value="views">最多浏览</option>
        </select>
      </div>

      {/* 重置按钮 */}
      <button onClick={() => {
        setSelectedCategories([]);
        setSelectedTags([]);
        setDateRange({ start: '', end: '' });
      }}>
        重置筛选
      </button>
    </div>
  );
}
```
