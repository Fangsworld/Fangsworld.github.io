## 1. CRUD 操作

### Create（创建）

```typescript
import { db } from '@/db';
import { articles } from '@/db/schema';

// 插入单条记录
const [newArticle] = await db.insert(articles)
  .values({
    slug: 'my-first-article',
    title: '我的第一篇文章',
    excerpt: '这是文章摘要',
    category: 'reading',
  })
  .returning();

// 结果：{ id: 1, slug: 'my-first-article', title: '我的第一篇文章', ... }

// 插入多条记录
const newArticles = await db.insert(articles)
  .values([
    { slug: 'article-1', title: '文章1', category: 'reading' },
    { slug: 'article-2', title: '文章2', category: 'writing' },
  ])
  .returning();
```

### Read（读取）

```typescript
// 查询所有记录
const allArticles = await db.select().from(articles);

// 查询特定字段
const articleTitles = await db.select({
  title: articles.title,
  slug: articles.slug,
})
.from(articles);

// 条件查询
const singleArticle = await db.select()
  .from(articles)
  .where(eq(articles.slug, 'my-first-article'));

// 多条件查询
import { and, or, gte, lte } from 'drizzle-orm';

const popularArticles = await db.select()
  .from(articles)
  .where(
    and(
      gte(articles.views, 100),
      lte(articles.views, 1000)
    )
  );

// 排序
import { desc } from 'drizzle-orm';

const topArticles = await db.select()
  .from(articles)
  .orderBy(desc(articles.views))
  .limit(10);
```

### Update（更新）

```typescript
// 更新单条记录
await db.update(articles)
  .set({ title: '新标题' })
  .where(eq(articles.slug, 'my-first-article'));

// 批量更新
await db.update(articles)
  .set({ views: sql`${articles.views} + 1` })
  .where(inArray(articles.slug, ['article-1', 'article-2']));

// 更新并返回
const [updated] = await db.update(articles)
  .set({ title: '更新后的标题' })
  .where(eq(articles.id, 1))
  .returning();
```

### Delete（删除）

```typescript
// 删除单条记录
await db.delete(articles)
  .where(eq(articles.slug, 'my-first-article'));

// 批量删除
await db.delete(articles)
  .where(eq(articles.category, 'reading'));
```

## 2. 复杂查询

### 连接查询（JOIN）

```typescript
import { sql } from 'drizzle-orm';

// 获取文章及其评论数
const articlesWithCommentCount = await db
  .select({
    id: articles.id,
    title: articles.title,
    commentCount: sql<number>`COUNT(${comments.id})`.mapWith(Number),
  })
  .from(articles)
  .leftJoin(comments, eq(articles.slug, comments.articleSlug))
  .groupBy(articles.id);

// 左连接
const result = await db
  .select({
    article: articles,
    comment: comments,
  })
  .from(articles)
  .leftJoin(comments, eq(articles.slug, comments.articleSlug));
```

### 聚合查询

```typescript
import { count, avg, sum } from 'drizzle-orm';

// 统计文章数
const totalArticles = await db
  .select({ count: count() })
  .from(articles);

// 按分类统计
const categoryStats = await db
  .select({
    category: articles.category,
    count: count(),
    totalViews: sum(articles.views).mapWith(Number),
  })
  .from(articles)
  .groupBy(articles.category);

// 平均浏览量
const avgViews = await db
  .select({ avg: avg(articles.views).mapWith(Number) })
  .from(articles);
```

### 子查询

```typescript
// 获取浏览量高于平均的文章
const avgViews = db
  .select({ avg: avg(articles.views) })
  .from(articles);

const aboveAverage = await db.select()
  .from(articles)
  .where(
    gt(
      articles.views,
      sql`(SELECT AVG(views) FROM articles)`
    )
  );
```

### 分页查询

```typescript
// OFFSET 分页
const page = 2;
const pageSize = 10;

const articles = await db.select()
  .from(articles)
  .limit(pageSize)
  .offset((page - 1) * pageSize);

// 游标分页（更高效）
const lastId = 100;
const articles = await db.select()
  .from(articles)
  .where(lt(articles.id, lastId))
  .orderBy(desc(articles.id))
  .limit(10);
```

## 3. 关系查询

### 使用 Query API

```typescript
// 获取文章及其评论
const articleWithComments = await db.query.articles.findFirst({
  where: eq(articles.slug, 'my-article'),
  with: {
    comments: {
      where: eq(comments.status, 'approved'),
      orderBy: [desc(comments.createdAt)],

// 获取文章及点赞数
const articleWithLikes = await db.query.articles.findFirst({
  where: eq(articles.slug, 'my-article'),
```

## 5. 性能优化技巧

### 避免 N+1 查询

```typescript
// ❌ 错误：N+1 查询
const articleList = await db.select().from(articles);
for (const article of articleList) {
  article.comments = await db.select()
    .from(comments)
    .where(eq(comments.articleSlug, article.slug));
}
// 执行 1 + N 次查询

// ✅ 正确：批量查询
const articleList = await db.select().from(articles);
const allComments = await db.select()
  .from(comments)
  .where(inArray(comments.articleSlug, articleList.map(a => a.slug)));

const commentsMap = new Map();
allComments.forEach(c => {
  if (!commentsMap.has(c.articleSlug)) {
    commentsMap.set(c.articleSlug, []);
  }
  commentsMap.get(c.articleSlug).push(c);
});

articleList.forEach(article => {
  article.comments = commentsMap.get(article.slug) || [];
});
```

### 只查询需要的字段

```typescript
// ❌ 查询所有字段
const articles = await db.select().from(articles);

// ✅ 只查询需要的字段
const articleList = await db.select({
  id: articles.id,
  title: articles.title,
  slug: articles.slug,
}).from(articles);
```

### 使用 Prepared Statements

```typescript
// Drizzle 自动使用 prepared statements
// 但你可以显式创建以提高性能

const getArticle = db.query.articles.findFirst({
  where: eq(articles.slug, sql.placeholder('slug')),
});

// 多次调用时性能更好
await getArticle.execute({ slug: 'article-1' });
await getArticle.execute({ slug: 'article-2' });
```

## 6. 实际项目中的查询

### 获取文章列表

```typescript
// lib/articles.ts
export async function getArticlesList(options: {
  category?: string;
  limit?: number;
  offset?: number;
}) {
  const query = db.select()
    .from(articles)
    .orderBy(desc(articles.createdAt))
    .limit(options.limit || 10)
    .offset(options.offset || 0);

  if (options.category) {
    query.where(eq(articles.category, options.category));
  }

  return await query;
}
```

### 获取文章详情

```typescript
export async function getArticleBySlug(slug: string) {
  const article = await db.query.articles.findFirst({
    where: eq(articles.slug, slug),
    with: {
      comments: {
        where: eq(comments.status, 'approved'),
        orderBy: [desc(comments.createdAt)],
      },
      likes: true,
    },
  });

  return article;
}
```

### 搜索文章

```typescript
export async function searchArticles(query: string) {
  return await db.select()
    .from(articles)
    .where(
      or(
        ilike(articles.title, `%${query}%`),
        ilike(articles.excerpt, `%${query}%`)
      )
    );
}
```
